export class EnrollmentService {
  constructor(private readonly repo: EnrollmentRepository) {}

  async execute(dto: CreateEnrollmentDTO): Promise<EnrollmentResult> {
    const student = await this.repo.findActiveStudent(dto.studentId);
    return this.repo.transaction(async (tx) => tx.enroll(student.id, dto.courseId));
  }
}
